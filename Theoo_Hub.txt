local function New(class, props, parent)
	local obj=Instance.new(class)
	for p,v in pairs(props or {}) do obj[p]=v end
	if parent then obj.Parent=parent end
	return obj
end

local function Round(obj,radius)
	local c=Instance.new("UICorner")
	c.CornerRadius=UDim.new(0,radius or 8)
	c.Parent=obj
	return c
end

local function Stroke(obj,color,thickness)
	local s=Instance.new("UIStroke")
	s.Color=color
	s.Thickness=thickness or 1
	s.Parent=obj
	return s
end

local function Tween(obj,props,duration)
	if not Config.UIAnimations then
		for p,v in pairs(props) do obj[p]=v end
		return
	end
	TweenService:Create(obj,TweenInfo.new(duration or .2,Enum.EasingStyle.Quad,Enum.EasingDirection.Out),props):Play()
end

local function RunPrerequisites()
	-- Refresh the camera reference before building UI/FOV systems.
	Camera=workspace.CurrentCamera
	while not Camera do
		task.wait()
		Camera=workspace.CurrentCamera
	end

	-- Make sure the local character and core humanoid objects are ready.
	local character=LocalPlayer.Character or LocalPlayer.CharacterAdded:Wait()
	character:WaitForChild("Humanoid")
	character:WaitForChild("HumanoidRootPart")

	-- Only prepare the objects required before menu initialization.
	task.wait()
	return true
end

		local percent=math.clamp((x-bar.AbsolutePosition.X)/math.max(bar.AbsoluteSize.X,1),0,1)
		local value=math.floor(min+(max-min)*percent+.5)
		fill.Size=UDim2.new(percent,0,1,0); valueLabel.Text=tostring(value)

	UIS.InputEnded:Connect(function(input)
		if input.UserInputType==Enum.UserInputType.MouseButton1 or input.UserInputType==Enum.UserInputType.Touch then dragging=false end

local TEAM_MARKERS = {
	"StageTeamSide", "Team", "TeamName", "TeamColor",
	"TeamId", "TeamID", "team"
}

local function ReadTeamMarker(inst)
	if not inst then return nil end
	for _,name in ipairs(TEAM_MARKERS) do
		local attr=inst:GetAttribute(name)
		if attr~=nil then
			return name.."="..tostring(attr)
		end
		local child=inst:FindFirstChild(name)
		if child and child:IsA("ValueBase") then
			return name.."="..tostring(child.Value)
		end
	end
	return nil
end

local function GetCustomTeamIdentity(player)
	if not player then return nil end
	local marker=ReadTeamMarker(player) or ReadTeamMarker(player.Character)
	if marker then return marker end
	local stats=player:FindFirstChild("leaderstats")
	marker=ReadTeamMarker(stats)
	if marker then return marker end
	if not player.Neutral then
		return "TeamColor="..tostring(player.TeamColor)
	end
	return nil
end

local customTeamCache={count=0,at=0}

local function GetCustomTeamCount()
	local now=os.clock()
	if now-customTeamCache.at>0.5 then
		local seen={}
		local count=0
		for _,player in ipairs(Players:GetPlayers()) do
			local identity=GetCustomTeamIdentity(player)
			if identity and not seen[identity] then
				seen[identity]=true
				count=count+1
			end
		end
		customTeamCache.count=count
		customTeamCache.at=now
	end
	return customTeamCache.count
end

local function GetTeamKey(player)
	if player and player.Team then
		return "team:"..player.Team.Name
	end
	return GetCustomTeamIdentity(player)
end

local function IsTeammate(player)
	if not player or player==LocalPlayer then return false end

	-- Prefer Roblox's native Team system when both players have a Team.
	if LocalPlayer.Team and player.Team then
		return LocalPlayer.Team==player.Team
	end

	local localTeam=GetCustomTeamIdentity(LocalPlayer)
	local targetTeam=GetCustomTeamIdentity(player)
	if not localTeam or not targetTeam then return false end

	-- Only trust custom markers when at least two distinct custom teams exist.
	return GetCustomTeamCount()>=2 and localTeam==targetTeam
end

local function IsEnemy(player)
	if player==LocalPlayer then return false end
	if Config.TeamCheck and IsTeammate(player) then return false end
	return true
end

local function GetCharacter(player)
	local c=player and player.Character
	if not c or not c.Parent then return nil end
	local h=c:FindFirstChildOfClass("Humanoid")
	if not h or h.Health<=0 then return nil end
	return c
end

local function GetRoot(c)
	return c:FindFirstChild("HumanoidRootPart") or c:FindFirstChild("UpperTorso") or c:FindFirstChild("Torso")
end

RemoveESP=function(player)
	local d=ESPObjects[player]
	if not d then return end
	if d.Highlight then d.Highlight:Destroy() end
	if d.Billboard then d.Billboard:Destroy() end
	ESPObjects[player]=nil
end

local function CreateESP(player)
	local c=GetCharacter(player)
	if player==LocalPlayer or not c then return end
	RemoveESP(player)
	local root=GetRoot(c)
	if not root then return end
	local h=Instance.new("Highlight")
	h.Name="THEOO_HUB_ESP_Highlight"; h.Adornee=c; h.FillColor=Theme.Accent; h.OutlineColor=Theme.Accent
	h.FillTransparency=.75; h.DepthMode=Enum.HighlightDepthMode.AlwaysOnTop
	h.Enabled=Config.ESPEnabled and Config.ShowHighlight; h.Parent=c
	local bb=Instance.new("BillboardGui")
	bb.Name="THEOO_HUB_ESP_Info"; bb.Adornee=root; bb.Size=UDim2.fromOffset(200,70); bb.StudsOffset=Vector3.new(0,3,0); bb.AlwaysOnTop=true; bb.Enabled=Config.ESPEnabled; bb.Parent=c
	local t=Instance.new("TextLabel")
	t.Size=UDim2.fromScale(1,1); t.BackgroundTransparency=1; t.TextColor3=Color3.new(1,1,1); t.TextStrokeTransparency=0
	t.TextSize=13; t.Font=Enum.Font.GothamBold; t.TextWrapped=true; t.Parent=bb
	ESPObjects[player]={Character=c,Highlight=h,Billboard=bb,Text=t}
end

local function UpdateESP()
	for _,p in ipairs(Players:GetPlayers()) do
		if p~=LocalPlayer then
			local c=GetCharacter(p)
			if not Config.ESPEnabled or not IsEnemy(p) or not c then RemoveESP(p)
			else
				local d=ESPObjects[p]
				if not d or d.Character~=c then CreateESP(p); d=ESPObjects[p] end
				if d then
					local root=GetRoot(c); local h=c:FindFirstChildOfClass("Humanoid")
					if root and h then
						d.Highlight.Enabled=Config.ESPEnabled and Config.ShowHighlight
						d.Billboard.Enabled=Config.ESPEnabled
						local lines={}
						if Config.ShowNames then table.insert(lines,p.DisplayName) end
						if Config.ShowHealth then table.insert(lines,"HP: "..math.floor(h.Health)) end
						if Config.ShowDistance then table.insert(lines,math.floor((Camera.CFrame.Position-root.Position).Magnitude).." studs") end
						d.Text.Text=table.concat(lines,"\n")
					end
				end
			end
		end
	end
end

local DummyFolder=workspace:FindFirstChild("TrainingDummies")
local function ApplyDummySettings()
	if not DummyFolder then return end
	for _,dummy in ipairs(DummyFolder:GetChildren()) do
		if dummy:IsA("Model") then
			local h=dummy:FindFirstChildOfClass("Humanoid")
			if h then
				for _,part in ipairs(dummy:GetDescendants()) do
					if part:IsA("BasePart") then
						if not part:GetAttribute("SNUG_OriginalSize") then part:SetAttribute("SNUG_OriginalSize",part.Size) end
						local original=part:GetAttribute("SNUG_OriginalSize")
						part.Size=Config.TrainingHitbox and original*2 or original
					end
				end
				local head=dummy:FindFirstChild("Head")
				if head and head:IsA("BasePart") then
					if not head:GetAttribute("SNUG_OriginalHeadSize") then head:SetAttribute("SNUG_OriginalHeadSize",head.Size) end
					local original=head:GetAttribute("SNUG_OriginalHeadSize")
					head.Size=Config.BigHead and original*2 or original
				end
			end
		end
	end
end

local TotalDamage,HitCount,DamageStart=0,0,os.clock()
local function RegisterDamage(amount)
	if amount<=0 then return end
	TotalDamage+=amount; HitCount+=1
	local dps=TotalDamage/math.max(os.clock()-DamageStart,.1)
	DamageText.Text="Damage: "..math.floor(TotalDamage); HitText.Text="Hits: "..HitCount; DPSText.Text="DPS: "..math.floor(dps); CounterFrame.Visible=true
end
local function MonitorDummy(model)
	if not model:IsA("Model") then return end
	local h=model:FindFirstChildOfClass("Humanoid")
	if not h or h:GetAttribute("SNUG_DamageMonitor") then return end
	h:SetAttribute("SNUG_DamageMonitor",true)
	local last=h.Health
	h.HealthChanged:Connect(function(n) if n<last then RegisterDamage(last-n) end; last=n end)
end

local function GetAimPart(c)
	if not c then return nil end
	if Config.AimTarget=="Head" then return c:FindFirstChild("Head")
	elseif Config.AimTarget=="Torso" then return c:FindFirstChild("UpperTorso") or c:FindFirstChild("Torso") or c:FindFirstChild("HumanoidRootPart")
	else return c:FindFirstChild("LeftLowerLeg") or c:FindFirstChild("RightLowerLeg") or c:FindFirstChild("Left Leg") or c:FindFirstChild("Right Leg") or c:FindFirstChild("HumanoidRootPart") end
end

local function CanSeeTarget(part,c)
	if not Config.WallCheck then return true end
	local params=RaycastParams.new()
	params.FilterType=Enum.RaycastFilterType.Exclude
	params.FilterDescendantsInstances=LocalPlayer.Character and {LocalPlayer.Character} or {}
	local result=workspace:Raycast(Camera.CFrame.Position,part.Position-Camera.CFrame.Position,params)
	return not result or result.Instance:IsDescendantOf(c)
end

local function GetFOVViewportCenter()
	-- Coordinate system used by Camera:WorldToViewportPoint().
	local viewport=Camera.ViewportSize
	return Vector2.new(viewport.X*0.5,viewport.Y*0.5)
end

local function GetFOVScreenCenter()
	-- Coordinate system used by our IgnoreGuiInset=true ScreenGui.
	-- The inset is added only for drawing the circle, not for target math.
	local viewport=Camera.ViewportSize
	-- THEOO_HUB uses IgnoreGuiInset=true, so its GUI coordinate system
	-- must use the same raw viewport center as WorldToViewportPoint().
	return Vector2.new(viewport.X*0.5,viewport.Y*0.5)
end

local function GetFOVRadius()
	local viewport=Camera.ViewportSize
	local shortestSide=math.min(viewport.X,viewport.Y)
	local percent=math.clamp(tonumber(Config.FOVPercent) or 1,1,100)
	local minRadius=2
	local maxRadius=shortestSide*0.35
	local t=(percent-1)/99
	return minRadius+(maxRadius-minRadius)*t
end

local function GetClosestTarget()
	local center=GetFOVViewportCenter()
	local radius=GetFOVRadius()
	local closest,dist=nil,math.huge
	for _,p in ipairs(Players:GetPlayers()) do
		if p~=LocalPlayer and IsEnemy(p) then
			local c=GetCharacter(p); local part=GetAimPart(c)
			if part then
				local pos,on=Camera:WorldToViewportPoint(part.Position)
				if on then
					local d=(Vector2.new(pos.X,pos.Y)-center).Magnitude
					if d<=radius and d<dist and CanSeeTarget(part,c) then closest,dist=part,d end
				end
			end
		end
	end
	return closest
end

-- Prevent duplicate binding on re-execution.
pcall(function() RunService:UnbindFromRenderStep("THEOO_HUB_AimLoop") end)
RunService:BindToRenderStep("THEOO_HUB_AimLoop",Enum.RenderPriority.Camera.Value+1,function()
	if not Gui.Enabled or not Config.AimEnabled then return end
	local target=GetClosestTarget()
	if target then
		local cf=CFrame.lookAt(Camera.CFrame.Position,target.Position)
		Camera.CFrame=Camera.CFrame:Lerp(cf,math.clamp(Config.AimStrength/100,.01,1))
	end
end)
